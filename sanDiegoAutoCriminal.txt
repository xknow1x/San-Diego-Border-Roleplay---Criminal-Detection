local _hasGetMouseLocation = pcall(function() UserInputService:GetMouseLocation() end)
local function getMousePos()
    if _hasGetMouseLocation then
        return UserInputService:GetMouseLocation()
    end
    if mouse then return Vector2.new(mouse.X, mouse.Y) end
    return Vector2.zero
end

local _hasIsmouse1pressed = type(ismouse1pressed) == "function"
local function isLeftMouseDown()
    if _hasIsmouse1pressed then
        local ok, v = pcall(ismouse1pressed)
        if ok then return v end
    end
    local ok, v = pcall(function()
        return UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1)
    end)
    if ok then return v end
    return false
end

local WorldToScreen = WorldToScreen or function(pos)
    local cam = workspace.CurrentCamera
    if cam then
        local sp, on = cam:WorldToViewportPoint(pos)
        return Vector2.new(sp.X, sp.Y), on
    end
    return Vector2.zero, false
end

    end)
end

-- =====================================================================
-- BOAT DETECTION (CollectionService tag "Boat" from BoatUtil)
-- =====================================================================
local BOAT_TAG = "Boat"

local function isInBoat(char)
    if not char then return false end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum or not hum.Sit then return false end
    local seat = hum.SeatPart
    if not seat then return false end
    local p = seat.Parent
    while p do
        if CollectionService:HasTag(p, BOAT_TAG) then return true end
        p = p.Parent
    end
    return false
end

-- STRICT allowlist: ONLY players explicitly on a Civilian team are opponents.
-- No team / unknown team = NOT a civilian (treated as non-threat, excluded from list).
local function isCivilian(plr)
    local team = plr.Team
    if not team then return false end
    local n = team.Name:lower()
    return n == "civilian" or n == "civilians"
        or n:find("civilian") ~= nil
end

-- Blocklist: every law-enforcement / government / allied team.
-- Coast Guard, US Army, FBI, BORTAC, Border Patrol, Police, SWAT are all teammates.
local function isLawEnforcement(plr)
    local team = plr.Team
    if not team then return false end
    local n = team.Name:lower()
    return n:find("police")       ~= nil
        or n:find("bortac")       ~= nil
        or n:find("patrol")       ~= nil  -- Border Patrol, USBP
        or n:find("border")       ~= nil
        or n:find("fbi")          ~= nil
        or n:find("swat")         ~= nil
        or n:find("sheriff")      ~= nil
        or n:find("coast guard")  ~= nil
        or n:find("coastguard")   ~= nil
        or n:find("coast")        ~= nil
        or n:find("guard")        ~= nil
        or n:find("army")         ~= nil
        or n:find("military")     ~= nil
        or n:find("national")     ~= nil  -- National Guard
        or n:find("us army")      ~= nil
        or n:find("srt")          ~= nil
        or n:find("fire")         ~= nil
        or n:find("ems")          ~= nil
        or n:find("paramedic")    ~= nil
        or n:find("dot")          ~= nil
        or n:find("federal")      ~= nil
        or n:find("ranger")       ~= nil
        or n:find("marshal")      ~= nil
end

-- =====================================================================
-- NEAREST SUSPECT (from cached refs)
-- =====================================================================
local function getNearestUsSuspect()
    local myChar = lp.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil, math.huge end
    local myPos = myRoot.Position
    local nearest, shortest = nil, math.huge
    for _, plr in ipairs(cachedSuspectRefs) do
        local char = plr.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if root then
            local d = (root.Position - myPos).Magnitude
            if d < shortest then shortest = d; nearest = plr end
        end
    end
    return nearest, shortest
end

task.spawn(function()
    while running do
        local espList     = {}
        local suspectData = {}
        local suspectRefs = {}
        local aptOwnerIds = {}

        local myChar = lp.Character
        local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
        local myPos  = myRoot and myRoot.Position or Vector3.zero

        local playerList = Players:GetPlayers()

        -- Pass 1: apartment owner set
        for i = 1, #playerList do
            local plr = playerList[i]
            if plr ~= lp then
                local aid = plr:GetAttribute("OwnedApartmentId")
                if aid then aptOwnerIds[plr.UserId] = true end
            end
        end

        local aptLook  = apartmentLookEnabled
        local boatLook = boatSmugglingEnabled

        -- Pass 2: main scan
        for i = 1, #playerList do
            local plr = playerList[i]
            if plr == lp then continue end

            local isCiv = isCivilian(plr)
            local isCop = isLawEnforcement(plr)

            local wantedLevel  = plr:GetAttribute("WantedLevel") or 0
            local stars        = math.clamp(math.ceil(wantedLevel / 200), 0, 5)
            local fromMexico   = plr:GetAttribute("FromMexico") == true
            local legallyCrossed = plr:GetAttribute("LegallyCrossedBorder") == true
            local granted      = plr:GetAttribute("Granted") == true
            local isIllegal    = fromMexico and not legallyCrossed and not granted
            local ownsApt      = aptOwnerIds[plr.UserId] == true

            local char     = plr.Character
            local rootPart = char and char:FindFirstChild("HumanoidRootPart")
            local isInUS   = rootPart and (rootPart.Position.X < BORDER_X) or false

            -- Boat smuggler = civilian, in US, seated in a Boat model, AND flagged
            local inBoatNow      = isInUS and isInBoat(char)
            local isBoatSmuggler = inBoatNow and isCiv and (isIllegal or wantedLevel > 0)

            local tName = plr.Team and plr.Team.Name or "Civilian"

            -- ── CIVILIAN-ONLY LIST ──
            if isCiv and not isCop and isInUS then
                local include = false
                if boatLook then
                    include = isBoatSmuggler
                elseif aptLook then
                    include = ownsApt and (wantedLevel > 0 or isIllegal)
                else
                    include = wantedLevel > 0 or isIllegal
                end

                if include then
                    suspectData[#suspectData + 1] = {
                        plr         = plr,
                        wantedLevel = wantedLevel,
                        stars       = stars,
                        tName       = tName,
                        ownsApt     = ownsApt,
                        isIllegal   = isIllegal,
                        isBoat      = isBoatSmuggler,
                        rootRef     = rootPart,
                    }
                    suspectRefs[#suspectRefs + 1] = plr
                end
            end

            -- ── ESP (all players in scope, not just civilians) ──
            if char then
                local showESP = false
                if boatLook then
                    showESP = isBoatSmuggler
                elseif aptLook then
                    showESP = isCiv and ownsApt and not isCop and (wantedLevel > 0 or isIllegal)
                else
                    showESP = isInUS
                end

                if showESP then
                    local head = char:FindFirstChild("Head")
                    local hum  = char:FindFirstChildOfClass("Humanoid")
                    if head and rootPart and hum and hum.Health > 0 then
                        local dist   = math.floor((rootPart.Position - myPos).Magnitude)
                        local aptTag = ownsApt and " [APT]" or ""
                        local tagText, tagColor

                        if not isCiv then
                            -- Teammate (cop, coast guard, army, etc.) - plain label only
                            tagColor = Color3.fromRGB(80, 160, 255)
                            tagText  = ("[%s] %s (%dm)"):format(tName:upper(), plr.Name, dist)
                        elseif isBoatSmuggler then
                            tagColor = CLR_ESP_CYAN
                            if wantedLevel > 0 then
                                tagText = ("[BOAT-SMUG] [*%d] %s (%d pts) (%dm)%s"):format(stars, plr.Name, wantedLevel, dist, aptTag)
                            else
                                tagText = ("[BOAT-SMUG] %s (%dm) - Check Tablet%s"):format(plr.Name, dist, aptTag)
                            end
                        elseif wantedLevel > 0 then
                            tagColor = CLR_ESP_RED
                            tagText  = ("[CIV] [*%d] %s (%d pts) (%dm)%s"):format(stars, plr.Name, wantedLevel, dist, aptTag)
                        elseif isIllegal and isInUS then
                            tagColor = CLR_ESP_ORANGE
                            tagText  = ("[CIV] [SEARCH TABLET] %s (%dm)%s"):format(plr.Name, dist, aptTag)
                        elseif ownsApt and aptLook then
                            tagColor = CLR_ESP_PURPLE
                            tagText  = ("[CIV] [APT] [SEARCH TABLET] %s (%dm)"):format(plr.Name, dist)
                        else
                            tagColor = CLR_ESP_WHITE
                            tagText  = ("[CIV] %s (%dm)%s"):format(plr.Name, dist, aptTag)
                        end


                        espList[#espList + 1] = {
                            head    = head,
                            root    = rootPart,
                            tagText = tagText,
                            color   = tagColor,
                        }
                    end
                end
            end
        end

        -- Sort: wanted desc then distance asc
        if myRoot then
            table.sort(suspectData, function(a, b)
                if a.wantedLevel ~= b.wantedLevel then
                    return a.wantedLevel > b.wantedLevel
                end
                local da = a.rootRef and (a.rootRef.Position - myPos).Magnitude or math.huge
                local db = b.rootRef and (b.rootRef.Position - myPos).Magnitude or math.huge
                return da < db
            end)
        end

        cachedTargets        = espList
        cachedSuspectPlayers = suspectData
        cachedSuspectRefs    = suspectRefs
        cachedAptOwners      = aptOwnerIds
        task.wait(0.33)
    end
end)

-- =====================================================================
-- RENDER & INTERACTION LOOP
-- =====================================================================
RunService.RenderStepped:Connect(function()
    if not running then return end

    local mPos       = getMousePos()
    local isMouseDown = isLeftMouseDown()

    local suspectData   = cachedSuspectPlayers
    local totalSuspects = #suspectData
    local maxOffset     = math.max(0, totalSuspects - MAX_ROWS)

    -- CLICK HANDLING
    if isMouseDown and not lastClickState then
        if isInside(windowPos, Vector2.new(windowSize.X, 34), mPos) then
            isDragging = true
            dragOffset = Vector2.new(mPos.X - windowPos.X, mPos.Y - windowPos.Y)
        else
            if isInside(btnScrollUpBg.Position, btnScrollUpBg.Size, mPos) then
                scrollOffset = math.clamp(scrollOffset - 1, 0, maxOffset)
            end
            if isInside(btnScrollDnBg.Position, btnScrollDnBg.Size, mPos) then
                scrollOffset = math.clamp(scrollOffset + 1, 0, maxOffset)
            end

            for i = 1, MAX_ROWS do
                local row = playerRows[i]
                if row.player and isInside(row.bg.Position, row.bg.Size, mPos) then
                    selectedPlayer = row.player
                    if setclipboard then
                        setclipboard(row.player.Name)
                        notify("Target Locked", "Selected & Copied '" .. row.player.Name .. "'  Paste in Tablet!", 2)
                    else
                        notify("Target Locked", "Selected: " .. row.player.Name, 2)
                    end
                    break
                end
            end

            if isInside(btnCopyBg.Position, btnCopyBg.Size, mPos) then
                if selectedPlayer then
                    if setclipboard then
                        setclipboard(selectedPlayer.Name)
                        notify("Tablet Helper", "Copied '" .. selectedPlayer.Name .. "'  Ctrl+V in Tablet", 3)
                    end
                else
                    notify("Warning", "Select a suspect row first!", 2)
                end
            end

            if isInside(btnNearestBg.Position, btnNearestBg.Size, mPos) then
                local nearPlr, nearDist = getNearestUsSuspect()
                if nearPlr then
                    selectedPlayer = nearPlr
                    if setclipboard then setclipboard(nearPlr.Name) end
                    notify("Nearest Target", "Selected '" .. nearPlr.Name .. "' (" .. math.floor(nearDist) .. "m)", 3)
                else
                    notify("Nearest Target", "No suspects on US side!", 2)
                end
            end

            if isInside(btnUnstuckBg.Position, btnUnstuckBg.Size, mPos) then
                task.spawn(function()
                    local rem = ReplicatedStorage:FindFirstChild("__remotes")
                    if rem and rem:FindFirstChild("VehicleService") and rem.VehicleService:FindFirstChild("UnstuckVehicle") then
                        rem.VehicleService.UnstuckVehicle:FireServer()
                        notify("Vehicle", "Unstuck request sent!", 2)
                    end
                end)
            end

            if isInside(btnAptBg.Position, btnAptBg.Size, mPos) then
                if boatSmugglingEnabled then boatSmugglingEnabled = false end
                apartmentLookEnabled = not apartmentLookEnabled
                scrollOffset = 0
                notify("Apt Look", apartmentLookEnabled and "ENABLED - Apartment suspects" or "DISABLED", 3)
            end

            if isInside(btnBoatBg.Position, btnBoatBg.Size, mPos) then
                if apartmentLookEnabled then apartmentLookEnabled = false end
                boatSmugglingEnabled = not boatSmugglingEnabled
                scrollOffset = 0
                notify("Boat Smuggling", boatSmugglingEnabled and "ENABLED - Tracking boat smugglers!" or "DISABLED", 3)
            end
        end
    elseif not isMouseDown then
        isDragging = false
    end

    if isDragging then
        windowPos = Vector2.new(mPos.X - dragOffset.X, mPos.Y - dragOffset.Y)
    end

    lastClickState = isMouseDown

    if windowPos.X ~= lastWindowPos.X or windowPos.Y ~= lastWindowPos.Y then
        repositionPanel()
        lastWindowPos = windowPos
    end

    -- STATUS LABEL
    if selectedPlayer then
        local char  = selectedPlayer.Character
        local root  = char and char:FindFirstChild("HumanoidRootPart")
        local myC   = lp.Character
        local myR   = myC and myC:FindFirstChild("HumanoidRootPart")
        if root and myR then
            local d = math.floor((root.Position - myR.Position).Magnitude)
            statusLabel.Text = ("Target: %s  |  %dm  |  Ctrl+V in Tablet"):format(selectedPlayer.Name, d)
        else
            statusLabel.Text = ("Target: %s  |  Ctrl+V in Tablet"):format(selectedPlayer.Name)
        end
        statusLabel.Color = CLR_STATUS_GOLD
    else
        statusLabel.Text  = "Target: None  --  Click a suspect row to select & copy"
        statusLabel.Color = CLR_STATUS_DIM
    end

    -- RANGE LABEL
    scrollOffset = math.clamp(scrollOffset, 0, maxOffset)
    local startRange = (totalSuspects > 0) and (scrollOffset + 1) or 0
    local endRange   = math.min(scrollOffset + MAX_ROWS, totalSuspects)

    if boatSmugglingEnabled then
        listRangeLabel.Text  = ("BOAT SMUGGLERS US (%d-%d of %d)"):format(startRange, endRange, totalSuspects)
        listRangeLabel.Color = CLR_LABEL_BOAT
    elseif apartmentLookEnabled then
        listRangeLabel.Text  = ("APT SUSPECTS US (%d-%d of %d)"):format(startRange, endRange, totalSuspects)
        listRangeLabel.Color = CLR_LABEL_APT
    else
        listRangeLabel.Text  = ("Civilians US (%d-%d of %d)"):format(startRange, endRange, totalSuspects)
        listRangeLabel.Color = CLR_LABEL_NORMAL
    end

    -- PLAYER ROWS
    local rowStartY = windowPos.Y + 85
    local myChar2 = lp.Character
    local myRoot2 = myChar2 and myChar2:FindFirstChild("HumanoidRootPart")
    local myPos2  = myRoot2 and myRoot2.Position or Vector3.zero

    for i = 1, MAX_ROWS do
        local row  = playerRows[i]
        local tb   = threatBars[i]
        local idx  = i + scrollOffset
        local data = suspectData[idx]

        if data then
            local plr = data.plr
            row.player = plr

            local rPos = Vector2.new(windowPos.X + 8, rowStartY + (i - 1) * 24)
            row.bg.Position   = rPos
            row.text.Position = Vector2.new(rPos.X + 12, rPos.Y + 5)
            tb.Position       = rPos
            tb.Size           = Vector2.new(3, 23)

            local wantedLevel = data.wantedLevel
            local stars       = data.stars
            local ownsApt     = data.ownsApt
            local isIllegal   = data.isIllegal
            local isBoat      = data.isBoat

            local dist = 0
            if data.rootRef and myRoot2 then
                dist = math.floor((data.rootRef.Position - myPos2).Magnitude)
            end

            -- threat bar
            if isBoat then
                tb.Color = CLR_ESP_CYAN
            elseif wantedLevel > 0 then
                tb.Color = CLR_ESP_RED
            elseif isIllegal then
                tb.Color = CLR_ESP_ORANGE
            elseif ownsApt then
                tb.Color = CLR_ESP_PURPLE
            else
                tb.Color = CLR_ESP_WHITE
            end
            tb.Visible = true

            local bgBase = (i % 2 == 0) and CLR_ROW_BG_ALT or CLR_ROW_BG
            local aptMark  = ownsApt and " [APT]" or ""
            local boatMark = isBoat and " [BOAT]" or ""

            if plr == selectedPlayer then
                row.bg.Color   = CLR_SELECTED_BG
                row.text.Color = CLR_SELECTED_TEXT
                if boatSmugglingEnabled then
                    if wantedLevel > 0 then
                        row.text.Text = ("> [CIV] %s (%dm) [BOAT]  |  *%d (%d pts)"):format(plr.Name, dist, stars, wantedLevel)
                    else
                        row.text.Text = ("> [CIV] %s (%dm) [BOAT]  |  CHECK TABLET"):format(plr.Name, dist)
                    end
                elseif apartmentLookEnabled then
                    if wantedLevel > 0 then
                        row.text.Text = ("> [CIV] %s (%dm) [APT]  |  *%d (%d pts)"):format(plr.Name, dist, stars, wantedLevel)
                    else
                        row.text.Text = ("> [CIV] %s (%dm) [APT]  |  SEARCH TABLET"):format(plr.Name, dist)
                    end
                else
                    row.text.Text = ("> [CIV] %s (%dm)%s%s  |  *%d (%d pts)"):format(plr.Name, dist, aptMark, boatMark, stars, wantedLevel)
                end
            else
                if isBoat then
                    row.bg.Color = CLR_BOAT_ROW_BG
                elseif ownsApt and apartmentLookEnabled then
                    row.bg.Color = CLR_APT_ROW_BG
                else
                    row.bg.Color = bgBase
                end

                if isBoat and wantedLevel > 0 then
                    row.text.Color = CLR_BOAT_WANTED_TEXT
                elseif isBoat then
                    row.text.Color = CLR_BOAT_SEARCH_TEXT
                elseif wantedLevel > 0 then
                    row.text.Color = CLR_WANTED_TEXT
                elseif ownsApt and apartmentLookEnabled then
                    row.text.Color = isIllegal and CLR_APT_WANTED_TEXT or CLR_APT_SEARCH_TEXT
                elseif isIllegal then
                    row.text.Color = CLR_ESP_ORANGE
                else
                    row.text.Color = Color3.fromRGB(200, 215, 235)
                end

                if boatSmugglingEnabled then
                    if wantedLevel > 0 then
                        row.text.Text = ("  [CIV] %s (%dm) [BOAT]  |  *%d (%d pts)"):format(plr.Name, dist, stars, wantedLevel)
                    else
                        row.text.Text = ("  [CIV] %s (%dm) [BOAT]  |  CHECK TABLET"):format(plr.Name, dist)
                    end
                elseif apartmentLookEnabled then
                    if wantedLevel > 0 then
                        row.text.Text = ("  [CIV] %s (%dm) [APT]  |  *%d (%d pts)"):format(plr.Name, dist, stars, wantedLevel)
                    else
                        row.text.Text = ("  [CIV] %s (%dm) [APT]  |  SEARCH TABLET"):format(plr.Name, dist)
                    end
                else
                    row.text.Text = ("  [CIV] %s (%dm)%s%s  |  *%d (%d pts)"):format(plr.Name, dist, aptMark, boatMark, stars, wantedLevel)
                end
            end

            row.bg.Visible   = true
            row.text.Visible = true
        else
            row.player = nil
            row.bg.Visible   = false
            row.text.Visible = false
            tb.Visible       = false
        end
    end

    -- FILTER BUTTON STATES
    if apartmentLookEnabled then
        btnAptBg.Color   = CLR_APT_BTN_ON
        btnAptText.Text  = "[APT LOOK: ON]  Apartment Suspects"
        btnAptText.Color = CLR_APT_BTN_TEXT_ON
    else
        btnAptBg.Color   = CLR_APT_BTN_OFF
        btnAptText.Text  = "[APT LOOK: OFF]  Apartment Suspects"
        btnAptText.Color = CLR_APT_BTN_TEXT_OFF
    end

    if boatSmugglingEnabled then
        btnBoatBg.Color   = CLR_BOAT_BTN_ON
        btnBoatText.Text  = "[BOAT SMUG: ON]  Boat Smugglers Active"
        btnBoatText.Color = CLR_BOAT_BTN_TEXT_ON
    else
        btnBoatBg.Color   = CLR_BOAT_BTN_OFF
        btnBoatText.Text  = "[BOAT SMUG: OFF]  Boat Smugglers"
        btnBoatText.Color = CLR_BOAT_BTN_TEXT_OFF
    end

    -- 3D ESP
    local targets  = cachedTargets
    local rendered = 0

    for i = 1, #targets do
        if rendered >= POOL_SIZE then break end
        local t = targets[i]

        local top, v1 = WorldToScreen(t.head.Position + Vector3.new(0, 0.7, 0))
        local bot, v2 = WorldToScreen(t.root.Position - Vector3.new(0, 3.2, 0))

        if v1 and v2 then
            rendered = rendered + 1
            local e = pool[rendered]
            local h = math.max(4, bot.Y - top.Y)
            local w = math.max(3, h * 0.5)

            e.box.Position = Vector2.new(top.X - w * 0.5, top.Y)
            e.box.Size     = Vector2.new(w, h)
            e.box.Color    = t.color
            e.tag.Text     = t.tagText
            e.tag.Color    = t.color
            e.tag.Position = Vector2.new(top.X, top.Y - 15)
            e.box.Visible  = true
            e.tag.Visible  = true
        end
    end

    for i = rendered + 1, POOL_SIZE do
        pool[i].box.Visible = false
        pool[i].tag.Visible = false
    end
end)

notify("SDBR Suite v2.0 Loaded", "Civilians only | Boat Smug | Apt Look | All features active.", 5)

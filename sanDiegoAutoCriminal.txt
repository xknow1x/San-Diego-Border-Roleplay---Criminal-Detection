-- Safe Mouse Location Helper
local function getMousePos()
    local ok, loc = pcall(function() return UserInputService:GetMouseLocation() end)
    if ok and loc and typeof(loc) == "Vector2" then
        return loc
    end
    if mouse then
        return Vector2_new(mouse.X, mouse.Y)
    end
    return Vector2_new(0, 0)
end

-- Safe Mouse Click State Helper
local function isLeftMouseDown()
    if ismouse1pressed then
        local ok, pressed = pcall(ismouse1pressed)
        if ok then return pressed end
    end
    local ok, pressed = pcall(function()
        return UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1)
    end)
    if ok then return pressed end
    return false
end

-- Safe WorldToScreen fallback
local WorldToScreen = WorldToScreen or function(pos)
    local cam = workspace.CurrentCamera
    if cam and cam:IsA("Camera") then
        local screenPos, onScreen = cam:WorldToViewportPoint(pos)
        return Vector2_new(screenPos.X, screenPos.Y), onScreen
    end
    return Vector2_new(0, 0), false
end

-- Font Helper
local DEFAULT_FONT = (Drawing.Font and Drawing.Font.Plex) or 2

local function formatText(textObj, str, size, color, center, font)
    textObj.Text = str or ""
    textObj.Size = size or 12
    textObj.Color = color or Color3_fromRGB(255, 255, 255)
    textObj.Center = center or false
    textObj.Outline = true
    pcall(function() textObj.Font = font or DEFAULT_FONT end)
end

-- Safe Notification Helper
local function notify(title, text, duration)
    pcall(function()
        StarterGui:SetCore("SendNotification", {
            Title = title,
            Text = text,
            Duration = duration or 3
        })
    end)
end

local function getCustomTeamColor(team)
    if not team then return Color3_fromRGB(200, 200, 200) end
    local name = team.Name:lower()
    
    for key, color in pairs(TEAM_COLORS) do
        if string_find(name, key) then
            return color
        end
    end
    
    if team.TeamColor then
        return team.TeamColor.Color
    end
    return Color3_fromRGB(200, 200, 200)
end

local function isLawEnforcement(plr)
    if not plr or not plr.Parent then return false end
    local team = plr.Team
    if not team then return false end
    local name = team.Name:lower()
    return string_find(name, "police") or string_find(name, "bortac") or string_find(name, "patrol") or string_find(name, "fbi") or string_find(name, "swat") or string_find(name, "sheriff")
end

-- Helper: Lock onto Nearest US Suspect
local function getNearestUsSuspect()
    local myChar = lp.Character
    local myRoot = (myChar and myChar.Parent) and myChar:FindFirstChild("HumanoidRootPart")
    local myPos = myRoot and myRoot.Position or Vector3_zero
    
    local nearest = nil
    local shortestDist = math_huge
    
    for _, item in ipairs(cachedSuspectPlayers) do
        local plr = item.player
        if plr and plr.Parent and plr.Character and plr.Character.Parent then
            local root = plr.Character:FindFirstChild("HumanoidRootPart")
            if root then
                local dist = (root.Position - myPos).Magnitude
                if dist < shortestDist then
                    shortestDist = dist
                    nearest = plr
                end
            end
        end
    end
    return nearest, shortestDist
end

-- BACKGROUND COLLECTOR (3 Hz) - Pre-computes & Caches Calculations
-- ---------------------------------------------------------------------
task.spawn(function()
    while running do
        local espList = {}
        local suspectPlrs = {}
        local rosterText = ""
        local count = 0

        local myChar = lp.Character
        local myRoot = (myChar and myChar.Parent) and myChar:FindFirstChild("HumanoidRootPart")
        local myPos = myRoot and myRoot.Position or Vector3_zero

        for _, plr in ipairs(Players:GetPlayers()) do
            if plr and plr.Parent and plr ~= lp then
                local isCop = isLawEnforcement(plr)

                local wantedLevel = plr:GetAttribute("WantedLevel") or 0
                local stars = math_clamp(math_floor(wantedLevel / 200), 0, 5)
                local fromMexico = plr:GetAttribute("FromMexico") == true
                local legallyCrossed = plr:GetAttribute("LegallyCrossedBorder") == true
                local granted = plr:GetAttribute("Granted") == true
                local isIllegalCrosser = fromMexico and not legallyCrossed and not granted

                local char = plr.Character
                local rootPart = (char and char.Parent) and char:FindFirstChild("HumanoidRootPart")
                local isInsideUS = rootPart and (rootPart.Position.X < BORDER_US_X_LIMIT) or false
                local isDecayedCrosser = isIllegalCrosser and (wantedLevel == 0) and (plr.Team and plr.Team.Name == "Civilian")

                local dist = 0
                if rootPart then
                    dist = math_floor((rootPart.Position - myPos).Magnitude)
                end

                -- US Territory Suspect Filter
                if not isCop and isInsideUS and (wantedLevel > 0 or isIllegalCrosser) then
                    table_insert(suspectPlrs, {
                        player = plr,
                        wantedLevel = wantedLevel,
                        stars = stars,
                        dist = dist,
                        teamName = plr.Team and plr.Team.Name or "Civilian"
                    })
                end

                -- Build Roster HUD Text (US Side Only)
                if isInsideUS and (wantedLevel > 0 or isDecayedCrosser) then
                    count = count + 1
                    if count <= 12 then
                        if wantedLevel > 0 then
                            rosterText = rosterText .. string_format("[*%d] %s | %d pts | Team: %s\n", stars, plr.Name, wantedLevel, plr.Team and plr.Team.Name or "Civilian")
                        elseif isDecayedCrosser then
                            rosterText = rosterText .. string_format("[SEARCH TABLET] %s | Illegal Crosser (US)\n", plr.Name)
                        end
                    end
                end

                -- 3D ESP Targets (Displays Custom Colors per Team)
                if char and char.Parent then
                    local head = char:FindFirstChild("Head")
                    local root = char:FindFirstChild("HumanoidRootPart")
                    local hum = char:FindFirstChildOfClass("Humanoid")

                    if head and root and hum and hum.Health > 0 then
                        local teamName = plr.Team and plr.Team.Name or "No Team"
                        local teamColor = getCustomTeamColor(plr.Team)

                        local tagText = ""
                        local tagColor = teamColor
                        local upperTeam = string_upper(teamName)

                        if wantedLevel > 0 then
                            tagColor = Color3_fromRGB(255, 60, 60)
                            tagText = string_format("[%s] [*%d] %s (%d pts) - %dm", upperTeam, stars, plr.Name, wantedLevel, dist)
                        elseif isDecayedCrosser and isInsideUS then
                            tagColor = Color3_fromRGB(255, 170, 0)
                            tagText = string_format("[%s] [SEARCH TABLET] %s (%dm)", upperTeam, plr.Name, dist)
                        else
                            tagText = string_format("[%s] %s (%dm)", upperTeam, plr.Name, dist)
                        end

                        table_insert(espList, { head = head, root = root, tagText = tagText, color = tagColor })
                    end
                end
            end
        end

        if rosterText == "" then rosterText = "No Wanted Players currently active on US side." end
        hudList.Text = rosterText

        cachedTargets = espList
        cachedSuspectPlayers = suspectPlrs
        task.wait(0.33)
    end
end)

-- ---------------------------------------------------------------------
-- RENDER LOOP (Zero GC Memory Allocations for Maximum FPS)
-- ---------------------------------------------------------------------
RunService.RenderStepped:Connect(function()
    if not running then return end

    local mPos = getMousePos()
    local isMouseDown = isLeftMouseDown()

    -- Clean up destroyed target
    if selectedPlayer and (not selectedPlayer.Parent or not selectedPlayer:IsA("Player")) then
        selectedPlayer = nil
    end

    local suspectPlrs = cachedSuspectPlayers
    local totalSuspects = #suspectPlrs
    local maxOffset = math_max(0, totalSuspects - MAX_ROWS)

    -- Mouse Click & Drag Logic
    if isMouseDown and not lastClickState then
        if isInside(windowPos, Vector2_new(windowSize.X, 32), mPos) then
            isDragging = true
            dragOffset = Vector2_new(mPos.X - windowPos.X, mPos.Y - windowPos.Y)
        else
            -- Scroll UP Button Click
            if isInside(btnScrollUpBg.Position, btnScrollUpBg.Size, mPos) then
                scrollOffset = math_clamp(scrollOffset - 1, 0, maxOffset)
            end
            -- Scroll DOWN Button Click
            if isInside(btnScrollDnBg.Position, btnScrollDnBg.Size, mPos) then
                scrollOffset = math_clamp(scrollOffset + 1, 0, maxOffset)
            end

            -- Click Player Row (AUTO-SELECT + AUTO-COPY TO CLIPBOARD)
            for i = 1, MAX_ROWS do
                local row = playerRows[i]
                if row.player and row.player.Parent and isInside(row.bg.Position, row.bg.Size, mPos) then
                    selectedPlayer = row.player
                    if setclipboard then
                        setclipboard(row.player.Name)
                        notify("Target Selected", "Selected & Copied '" .. row.player.Name .. "' for Tablet!", 2)
                    else
                        notify("Target Selected", "Selected: " .. row.player.Name, 2)
                    end
                    break
                end
            end

            -- Action Button 1: Copy Selected Name
            if isInside(btnCopyBg.Position, btnCopyBg.Size, mPos) then
                if selectedPlayer and selectedPlayer.Parent then
                    if setclipboard then
                        setclipboard(selectedPlayer.Name)
                        notify("Tablet Helper", "Copied '" .. selectedPlayer.Name .. "'! Press Ctrl+V in Tablet.", 3)
                    end
                else
                    notify("Warning", "Click a suspect from the list first!", 2)
                end
            end

            -- Action Button 2: Lock onto Nearest Suspect
            if isInside(btnNearestBg.Position, btnNearestBg.Size, mPos) then
                local nearPlr, nearDist = getNearestUsSuspect()
                if nearPlr then
                    selectedPlayer = nearPlr
                    if setclipboard then setclipboard(nearPlr.Name) end
                    notify("Nearest Target", "Selected & Copied '" .. nearPlr.Name .. "' (" .. math_floor(nearDist) .. "m)!", 3)
                else
                    notify("Nearest Target", "No suspects found on US side!", 2)
                end
            end

            -- Action Button 3: Unstuck Vehicle
            if isInside(btnUnstuckBg.Position, btnUnstuckBg.Size, mPos) then
                task.spawn(function()
                    local remotes = ReplicatedStorage:FindFirstChild("__remotes")
                    if remotes and remotes:FindFirstChild("VehicleService") and remotes.VehicleService:FindFirstChild("UnstuckVehicle") then
                        remotes.VehicleService.UnstuckVehicle:FireServer()
                        notify("Vehicle", "Vehicle Unstuck Request Sent!", 2)
                    end
                end)
            end
        end
    elseif not isMouseDown then
        isDragging = false
    end

    if isDragging then
        windowPos = Vector2_new(mPos.X - dragOffset.X, mPos.Y - dragOffset.Y)
    end

    lastClickState = isMouseDown

    -- Position Frame Elements
    panelBg.Position = windowPos
    panelBorder.Position = windowPos
    panelHeader.Position = windowPos
    panelHeaderAccent.Position = windowPos
    panelTitle.Position = Vector2_new(windowPos.X + 12, windowPos.Y + 8)
    statusLabel.Position = Vector2_new(windowPos.X + 12, windowPos.Y + 38)

    local myChar = lp.Character
    local myRoot = (myChar and myChar.Parent) and myChar:FindFirstChild("HumanoidRootPart")
    local myPos = myRoot and myRoot.Position or Vector3_zero

    local selChar = selectedPlayer and selectedPlayer.Parent and selectedPlayer.Character
    local selRoot = (selChar and selChar.Parent) and selChar:FindFirstChild("HumanoidRootPart")

    if selectedPlayer and selRoot then
        local dist = math_floor((selRoot.Position - myPos).Magnitude)
        statusLabel.Text = string_format("Target: %s (%dm) - Copied! Press Ctrl+V in Tablet", selectedPlayer.Name, dist)
        statusLabel.Color = Color3_fromRGB(255, 215, 0)
    elseif selectedPlayer then
        statusLabel.Text = "Target: " .. selectedPlayer.Name .. " - Copied! Press Ctrl+V in Tablet"
        statusLabel.Color = Color3_fromRGB(255, 215, 0)
    else
        statusLabel.Text = "Target: None (Click suspect row below to select & copy)"
        statusLabel.Color = Color3_fromRGB(160, 175, 195)
    end

    scrollOffset = math_clamp(scrollOffset, 0, maxOffset)
    local startRange = (totalSuspects > 0) and (scrollOffset + 1) or 0
    local endRange = math.min(scrollOffset + MAX_ROWS, totalSuspects)

    local subHeaderY = windowPos.Y + 56
    listRangeLabel.Text = string_format("Suspects US (%d-%d of %d)", startRange, endRange, totalSuspects)
    listRangeLabel.Position = Vector2_new(windowPos.X + 12, subHeaderY + 2)

    -- Scroll Buttons Position
    btnScrollUpBg.Position = Vector2_new(windowPos.X + 224, subHeaderY)
    btnScrollUpText.Position = Vector2_new(windowPos.X + 251, subHeaderY + 4)

    btnScrollDnBg.Position = Vector2_new(windowPos.X + 284, subHeaderY)
    btnScrollDnText.Position = Vector2_new(windowPos.X + 311, subHeaderY + 4)

    -- Render Player Rows (Fast Cache Read)
    local startY = windowPos.Y + 82
    for i = 1, MAX_ROWS do
        local row = playerRows[i]
        local item = suspectPlrs[i + scrollOffset]
        local plr = item and item.player

        row.player = plr

        if item and plr and plr.Parent then
            local rPos = Vector2_new(windowPos.X + 8, startY + ((i - 1) * 26))
            row.bg.Position = rPos
            row.text.Position = Vector2_new(rPos.X + 8, rPos.Y + 5)

            if plr == selectedPlayer then
                row.bg.Color = Color3_fromRGB(45, 85, 135)
                row.text.Color = Color3_fromRGB(255, 235, 120)
                row.text.Text = string_format("> [%s] %s (%dm)  |  *%d (%d pts)", string_upper(item.teamName), plr.Name, item.dist, item.stars, item.wantedLevel)
            else
                row.bg.Color = Color3_fromRGB(24, 30, 40)
                row.text.Color = Color3_fromRGB(220, 225, 235)
                row.text.Text = string_format("  [%s] %s (%dm)  |  *%d (%d pts)", string_upper(item.teamName), plr.Name, item.dist, item.stars, item.wantedLevel)
            end

            row.bg.Visible = true
            row.text.Visible = true
        else
            row.bg.Visible = false
            row.text.Visible = false
        end
    end

    -- Render Action Buttons (Bottom Row)
    local btnY = windowPos.Y + windowSize.Y - 34
    
    btnCopyBg.Position = Vector2_new(windowPos.X + 8, btnY)
    btnCopyText.Position = Vector2_new(windowPos.X + 61, btnY + 7)

    btnNearestBg.Position = Vector2_new(windowPos.X + 122, btnY)
    btnNearestText.Position = Vector2_new(windowPos.X + 175, btnY + 7)

    btnUnstuckBg.Position = Vector2_new(windowPos.X + 236, btnY)
    btnUnstuckText.Position = Vector2_new(windowPos.X + 289, btnY + 7)

    -- Fast 3D ESP Rendering Loop
    local currentTargets = cachedTargets
    local rendered = 0

    for i = 1, #currentTargets do
        if rendered >= POOL_SIZE then break end
        local t = currentTargets[i]

        if t.head and t.head.Parent and t.root and t.root.Parent then
            local success1, top, v1 = pcall(function()
                return WorldToScreen(t.head.Position + Vector3_new(0, 0.7, 0))
            end)
            local success2, bot, v2 = pcall(function()
                return WorldToScreen(t.root.Position - Vector3_new(0, 3.2, 0))
            end)

            if success1 and success2 and v1 and v2 then
                rendered = rendered + 1
                local elem = pool[rendered]
                
                local h = math_max(4, bot.Y - top.Y)
                local w = math_max(3, h * 0.5)

                elem.box.Position = Vector2_new(top.X - w * 0.5, top.Y)
                elem.box.Size = Vector2_new(w, h)
                elem.box.Color = t.color

                elem.tag.Text = t.tagText
                elem.tag.Position = Vector2_new(top.X, top.Y - 14)

                elem.box.Visible = true
                elem.tag.Visible = true
            end
        end
    end

    for i = rendered + 1, POOL_SIZE do
        pool[i].box.Visible = false
        pool[i].tag.Visible = false
    end
end)

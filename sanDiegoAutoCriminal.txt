local _hasGetMouseLocation = pcall(function() UserInputService:GetMouseLocation() end)
local function getMousePos()
    if _hasGetMouseLocation then
        return UserInputService:GetMouseLocation()
    end
    if mouse then return Vector2.new(mouse.X, mouse.Y) end
    return Vector2.zero
end

local _hasIsmouse1pressed = type(ismouse1pressed) == "function"
local function isLeftMouseDown()
    if _hasIsmouse1pressed then
        local ok, v = pcall(ismouse1pressed)
        if ok then return v end
    end
    local ok, v = pcall(function()
        return UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1)
    end)
    if ok then return v end
    return false
end

    local cam = workspace.CurrentCamera

local BOAT_PROX_SQ = 20 * 20

local _boatsFolder = nil
local function getBoatsFolder()
    if _boatsFolder and _boatsFolder.Parent then return _boatsFolder end
    _boatsFolder = workspace:FindFirstChild("Boats")
    return _boatsFolder
end

local function isNearBoat(rootPart)
    if not rootPart then return false end
    local ok, result = pcall(function()
        local plrPos = rootPart.Position
        local folder = getBoatsFolder()
        if folder then
            for _, boat in ipairs(folder:GetChildren()) do
                local boatRoot = boat:FindFirstChild("Root") or boat.PrimaryPart or boat:FindFirstChildWhichIsA("BasePart")
                if boatRoot and boatRoot:IsA("BasePart") then
                    local diff = boatRoot.Position - plrPos
                    local distSq = diff.X * diff.X + diff.Y * diff.Y + diff.Z * diff.Z
                    if distSq <= BOAT_PROX_SQ then return true end
                end
            end
        end

        local char = rootPart.Parent
        if char and char.Parent and char.Parent ~= workspace then
            local pName = char.Parent.Name:lower()
            if pName:find("boat") or pName:find("dinghy") then
                return true
            end
        end
        return false
    end)
    return ok and result or false
end

local Teams = game:GetService("Teams")

local function getPlayerTeamName(plr)
    if not plr or plr.Parent ~= Players then return "Civilian" end
    local ok, name = pcall(function()
        if plr.Team then
            return plr.Team.Name
        end
        if plr.TeamColor then
            for _, t in ipairs(Teams:GetTeams()) do
                if t.TeamColor == plr.TeamColor then
                    return t.Name
                end
            end
        end
        return nil
    end)
    if ok and name then return name end
    return "Civilian"
end

local function isCivilian(plr)
    if not plr or plr.Parent ~= Players then return false end
    local tName = getPlayerTeamName(plr):lower()
    return tName == "civilian" or tName == "civilians" or tName:find("civilian") ~= nil
end

local function isLawEnforcement(plr)
    if not plr or plr.Parent ~= Players then return false end
    return not isCivilian(plr)
end

local function purgePlayer(plr)
    if not plr then return end
    for i = #cachedOfficerPlayers, 1, -1 do
        local entry = cachedOfficerPlayers[i]
        if not entry or entry.plr == plr or entry.plr.Parent ~= Players then
            table.remove(cachedOfficerPlayers, i)
        end
    end
    for i = #cachedOfficerRefs, 1, -1 do
        local ref = cachedOfficerRefs[i]
        if not ref or ref == plr or ref.Parent ~= Players then
            table.remove(cachedOfficerRefs, i)
        end
    end
    for i = #cachedSuspectPlayers, 1, -1 do
        local entry = cachedSuspectPlayers[i]
        if not entry or entry.plr == plr or entry.plr.Parent ~= Players then
            table.remove(cachedSuspectPlayers, i)
        end
    end
    for i = #cachedSuspectRefs, 1, -1 do
        local ref = cachedSuspectRefs[i]
        if not ref or ref == plr or ref.Parent ~= Players then
            table.remove(cachedSuspectRefs, i)
        end
    end
    if selectedPlayer == plr then
        selectedPlayer = nil
    end
    if myClosestOfficer == plr then
        myClosestOfficer = nil
        myClosestOfficerDist = math.huge
    end
end

pcall(function()

local function getNearestUsSuspect()
    local myChar = lp.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil, math.huge end
    local myPos = myRoot.Position
    local nearest, shortest = nil, math.huge
    for _, plr in ipairs(cachedSuspectRefs) do
        local char = plr.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if root then
            local d = (root.Position - myPos).Magnitude
            if d < shortest then shortest = d; nearest = plr end
        end
    end
    return nearest, shortest
end

                local fromMexico   = plr:GetAttribute("FromMexico") == true
                local legallyCrossed = plr:GetAttribute("LegallyCrossedBorder") == true
                local granted      = plr:GetAttribute("Granted") == true
                local isIllegal    = fromMexico and not legallyCrossed and not granted
                local ownsApt      = aptOwnerIds[plr.UserId] == true

                local char     = plr.Character
                local rootPart = char and char:FindFirstChild("HumanoidRootPart")
                local isInUS   = rootPart and (rootPart.Position.X < BORDER_X) or false
                local dist     = rootPart and math.floor((rootPart.Position - myPos).Magnitude) or 9999

                local inBoatNow      = isCiv and rootPart and isNearBoat(rootPart) or false
                local isBoatSmuggler = inBoatNow and isCiv

                local tName = getPlayerTeamName(plr)

                -- Collect Law Enforcement Officers
                if isCop then
                    officerData[#officerData + 1] = {
                        plr       = plr,
                        tName     = tName,
                        dist      = dist,
                        rootRef   = rootPart,
                    }
                    officerRefs[#officerRefs + 1] = plr
                end

                -- Collect Civilian Suspects
                if isCiv and not isCop and isInUS then
                    local include = false
                    if boatLook then
                        include = isBoatSmuggler
                    elseif aptLook then
                        include = ownsApt and (wantedLevel > 0 or isIllegal)
                    else
                        include = wantedLevel > 0 or isIllegal or isBoatSmuggler
                    end

                    if include then
                        suspectData[#suspectData + 1] = {
                            plr         = plr,
                            wantedLevel = wantedLevel,
                            stars       = stars,
                            tName       = tName,
                            ownsApt     = ownsApt,
                            isIllegal   = isIllegal,
                            isBoat      = isBoatSmuggler,
                            dist        = dist,
                            rootRef     = rootPart,
                        }
                        suspectRefs[#suspectRefs + 1] = plr
                    end
                end
            end

            -- Sort Officers by Distance ascending (closest officer at top)
            table.sort(officerData, function(a, b)
                return a.dist < b.dist
            end)

            local closestCop = officerData[1]
            myClosestOfficer     = closestCop and closestCop.plr or nil
            myClosestOfficerDist = closestCop and closestCop.dist or math.huge

            -- Sort Suspects by wanted desc then distance asc
            if myRoot then
                table.sort(suspectData, function(a, b)
                    if a.wantedLevel ~= b.wantedLevel then
                        return a.wantedLevel > b.wantedLevel
                    end
                    return a.dist < b.dist
                end)
            end

            -- Pass 3: Build ESP Targets
            for i = 1, #playerList do
                local plr = playerList[i]
                if not plr or plr.Parent ~= Players or plr == lp then continue end

                local char = plr.Character
                if char then
                    local isCiv = isCivilian(plr)
                    local isCop = isLawEnforcement(plr)
                    local rootPart = char:FindFirstChild("HumanoidRootPart")
                    local head = char:FindFirstChild("Head")
                    local hum  = char:FindFirstChildOfClass("Humanoid")
                    local hp   = (hum and hum.Health) or 100
                    local isInUS = rootPart and (rootPart.Position.X < BORDER_X) or false
                    local dist = rootPart and math.floor((rootPart.Position - myPos).Magnitude) or 0
                    local tName = getPlayerTeamName(plr)

                    local wantedLevel = plr:GetAttribute("WantedLevel") or 0
                    local stars = math.clamp(math.ceil(wantedLevel / 200), 0, 5)
                    local fromMexico = plr:GetAttribute("FromMexico") == true
                    local legallyCrossed = plr:GetAttribute("LegallyCrossedBorder") == true
                    local granted = plr:GetAttribute("Granted") == true
                    local isIllegal = fromMexico and not legallyCrossed and not granted
                    local ownsApt = aptOwnerIds[plr.UserId] == true
                    local isBoatSmuggler = isCiv and rootPart and isNearBoat(rootPart) or false
                    local aptTag = ownsApt and " [APT]" or ""

                    if head and rootPart and hp > 0 then
                        local showESP = false
                        local tagText, tagColor

                        if crimMode then
                            -- Criminal Mode ESP: Highlight ALL Officers, specially mark the closest one!
                            if isCop then
                                showESP = true
                                if plr == myClosestOfficer then
                                    tagColor = CLR_ESP_RED
                                    tagText  = ("[!] [CLOSEST LEO] [%s] %s (%dm)"):format(tName:upper(), plr.Name, dist)
                                else
                                    tagColor = CLR_ESP_BLUE
                                    tagText  = ("[LEO] [%s] %s (%dm)"):format(tName:upper(), plr.Name, dist)
                                end
                            end
                        else
                            -- Police Mode ESP: Standard Suspect / Teammate Highlights
                            if boatLook then
                                showESP = isBoatSmuggler
                            elseif aptLook then
                                showESP = isCiv and ownsApt and not isCop and (wantedLevel > 0 or isIllegal)
                            else
                                showESP = isInUS
                            end

                            if showESP then
                                if not isCiv then
                                    tagColor = Color3.fromRGB(80, 160, 255)
                                    tagText  = ("[%s] %s (%dm)"):format(tName:upper(), plr.Name, dist)
                                elseif isBoatSmuggler then
                                    tagColor = CLR_ESP_CYAN
                                    if wantedLevel > 0 then
                                        tagText = ("[BOAT-SMUG] [*%d] %s (%d pts) (%dm)%s"):format(stars, plr.Name, wantedLevel, dist, aptTag)
                                    else
                                        tagText = ("[BOAT-SMUG] %s (%dm) - Check Tablet%s"):format(plr.Name, dist, aptTag)
                                    end
                                elseif wantedLevel > 0 then
                                    tagColor = CLR_ESP_RED
                                    tagText  = ("[CIV] [*%d] %s (%d pts) (%dm)%s"):format(stars, plr.Name, wantedLevel, dist, aptTag)
                                elseif isIllegal and isInUS then
                                    tagColor = CLR_ESP_ORANGE
                                    tagText  = ("[CIV] [SEARCH TABLET] %s (%dm)%s"):format(plr.Name, dist, aptTag)
                                elseif ownsApt and aptLook then
                                    tagColor = CLR_ESP_PURPLE
                                    tagText  = ("[CIV] [APT] [SEARCH TABLET] %s (%dm)"):format(plr.Name, dist)
                                else
                                    tagColor = CLR_ESP_WHITE
                                    tagText  = ("[CIV] %s (%dm)%s"):format(plr.Name, dist, aptTag)
                                end
                            end
                        end

                        if showESP then
                            espList[#espList + 1] = {
                                head    = head,
                                root    = rootPart,
                                tagText = tagText,
                                color   = tagColor,
                            }
                        end
                    end
                end
            end

            cachedTargets        = espList
            cachedSuspectPlayers = suspectData
            cachedSuspectRefs    = suspectRefs
            cachedOfficerPlayers = officerData
            cachedOfficerRefs    = officerRefs
            cachedAptOwners      = aptOwnerIds
        end)
        task.wait(0.15)
    end
end)

        btnBoatText.Text  = "[BOAT SMUG: ON]  Boat Smugglers Active"
        btnBoatText.Color = CLR_BOAT_BTN_TEXT_ON
    else
        btnBoatBg.Color   = CLR_BOAT_BTN_OFF
        btnBoatText.Text  = "[BOAT SMUG: OFF]  Boat Smugglers"
        btnBoatText.Color = CLR_BOAT_BTN_TEXT_OFF
    end

    -- 3D ESP Rendering
    local targets  = cachedTargets
    local rendered = 0

    for i = 1, #targets do
        if rendered >= POOL_SIZE then break end
        local t = targets[i]

        local top, v1 = WorldToScreen(t.head.Position + Vector3.new(0, 0.7, 0))
        local bot, v2 = WorldToScreen(t.root.Position - Vector3.new(0, 3.2, 0))

        if v1 and v2 then
            rendered = rendered + 1
            local e = pool[rendered]
            local h = math.max(4, bot.Y - top.Y)
            local w = math.max(3, h * 0.5)

            e.box.Position = Vector2.new(top.X - w * 0.5, top.Y)
            e.box.Size     = Vector2.new(w, h)
            e.box.Color    = t.color
            e.tag.Text     = t.tagText
            e.tag.Color    = t.color
            e.tag.Position = Vector2.new(top.X, top.Y - 15)
            e.box.Visible  = true
            e.tag.Visible  = true
        end
    end

    for i = rendered + 1, POOL_SIZE do
        pool[i].box.Visible = false
        pool[i].tag.Visible = false
    end
end)

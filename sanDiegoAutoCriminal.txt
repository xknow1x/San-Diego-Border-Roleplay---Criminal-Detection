-- Safe Mouse Location Helper (optimized - cache method availability)
local _hasGetMouseLocation = pcall(function() UserInputService:GetMouseLocation() end)
local function getMousePos()
    if _hasGetMouseLocation then
        return UserInputService:GetMouseLocation()
    end
    if mouse then
        return Vector2.new(mouse.X, mouse.Y)
    end
    return Vector2.zero
end

-- Safe Mouse Click State Helper (optimized - cache method availability)
local _hasIsmouse1pressed = type(ismouse1pressed) == "function"
local function isLeftMouseDown()
    if _hasIsmouse1pressed then
        local ok, pressed = pcall(ismouse1pressed)
        if ok then return pressed end
    end
    local ok, pressed = pcall(function()
        return UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1)
    end)
    if ok then return pressed end
    return false
end

-- Safe WorldToScreen fallback
local WorldToScreen = WorldToScreen or function(pos)
    local cam = workspace.CurrentCamera
    if cam then
        local screenPos, onScreen = cam:WorldToViewportPoint(pos)
        return Vector2.new(screenPos.X, screenPos.Y), onScreen
    end
    return Vector2.zero, false
end

-- Font Helper
local DEFAULT_FONT = (Drawing.Font and Drawing.Font.Plex) or 2

local function formatText(textObj, str, size, color, center, font)
    textObj.Text = str or ""
    textObj.Size = size or 12
    textObj.Color = color or Color3.fromRGB(255, 255, 255)
    textObj.Center = center or false
    textObj.Outline = true
    pcall(function() textObj.Font = font or DEFAULT_FONT end)
end

-- Safe Notification Helper
local function notify(title, text, duration)
    pcall(function()
        game:GetService("StarterGui"):SetCore("SendNotification", {
            Title = title,
            Text = text,
            Duration = duration or 3
        })
    end)
end

local CLR_DEFAULT_TEAM = Color3.fromRGB(200, 200, 200)

local function getCustomTeamColor(team)
    if not team then return CLR_DEFAULT_TEAM end
    local name = team.Name:lower()
    for key, color in pairs(TEAM_COLORS) do
        if name:find(key) then
            return color
        end
    end
    if team.TeamColor then
        return team.TeamColor.Color
    end
    return CLR_DEFAULT_TEAM
end

local function isLawEnforcement(plr)
    local team = plr.Team
    if not team then return false end
    local name = team.Name:lower()
    return name:find("police") or name:find("bortac") or name:find("patrol") or name:find("fbi") or name:find("swat") or name:find("sheriff")
end

-- Helper: Lock onto Nearest US Suspect (uses cached refs)
local function getNearestUsSuspect()
    local myChar = lp.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil, math.huge end
    local myPos = myRoot.Position

    local nearest = nil
    local shortestDist = math.huge

    for _, plr in ipairs(cachedSuspectRefs) do
        local char = plr.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if root then
            local dist = (root.Position - myPos).Magnitude
            if dist < shortestDist then
                shortestDist = dist
                nearest = plr
            end
        end
    end
    return nearest, shortestDist
end

-- ---------------------------------------------------------------------
-- BACKGROUND COLLECTOR (3 Hz) - Single pass, pre-caches all row data
-- ---------------------------------------------------------------------
task.spawn(function()
    while running do
        local espList = {}
        local suspectData = {} -- Pre-computed row data: {plr, wantedLevel, stars, tName, teamColor, ownsApt, rootRef}
        local suspectRefs = {} -- Raw player refs for nearest-suspect
        local aptOwnerIdSet = {}

        local myChar = lp.Character
        local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
        local myPos = myRoot and myRoot.Position or Vector3.zero

        -- SINGLE PASS: build apt set + suspects + ESP in one loop
        local playerList = Players:GetPlayers()

        -- First quick pass: build apartment owner set (needed before main pass)
        for i = 1, #playerList do
            local plr = playerList[i]
            if plr ~= lp then
                local ownedAptId = plr:GetAttribute("OwnedApartmentId")
                if ownedAptId then
                    aptOwnerIdSet[plr.UserId] = true
                end
            end
        end

        -- Main pass: suspects + ESP
        local aptLook = apartmentLookEnabled -- Cache toggle state for this tick
        for i = 1, #playerList do
            local plr = playerList[i]
            if plr ~= lp then
                local isCop = isLawEnforcement(plr)
                local wantedLevel = plr:GetAttribute("WantedLevel") or 0
                local stars = math.clamp(math.ceil(wantedLevel / 200), 0, 5)
                local fromMexico = plr:GetAttribute("FromMexico") == true
                local legallyCrossed = plr:GetAttribute("LegallyCrossedBorder") == true
                local granted = plr:GetAttribute("Granted") == true
                local isIllegalCrosser = fromMexico and not legallyCrossed and not granted
                local ownsApartment = aptOwnerIdSet[plr.UserId] == true

                local char = plr.Character
                local rootPart = char and char:FindFirstChild("HumanoidRootPart")
                local isInsideUS = rootPart and (rootPart.Position.X < BORDER_US_X_LIMIT) or false

                local tName = plr.Team and plr.Team.Name or "Civilian"
                local teamColor = getCustomTeamColor(plr.Team)

                -- Suspect list filter
                if not isCop and isInsideUS then
                    if aptLook then
                        if ownsApartment and (wantedLevel > 0 or isIllegalCrosser) then
                            suspectData[#suspectData + 1] = {
                                plr = plr,
                                wantedLevel = wantedLevel,
                                stars = stars,
                                tName = tName,
                                teamColor = teamColor,
                                ownsApt = true,
                                rootRef = rootPart
                            }
                            suspectRefs[#suspectRefs + 1] = plr
                        end
                    else
                        if wantedLevel > 0 or isIllegalCrosser then
                            suspectData[#suspectData + 1] = {
                                plr = plr,
                                wantedLevel = wantedLevel,
                                stars = stars,
                                tName = tName,
                                teamColor = teamColor,
                                ownsApt = ownsApartment,
                                rootRef = rootPart
                            }
                            suspectRefs[#suspectRefs + 1] = plr
                        end
                    end
                end

                -- ESP filter
                local showESP = false
                if char then
                    if aptLook then
                        showESP = ownsApartment and not isCop and (wantedLevel > 0 or isIllegalCrosser)
                    else
                        showESP = isInsideUS
                    end
                end

                if showESP then
                    local head = char:FindFirstChild("Head")
                    local hum = char:FindFirstChildOfClass("Humanoid")

                    if head and rootPart and hum and hum.Health > 0 then
                        local dist = math.floor((rootPart.Position - myPos).Magnitude)
                        local tagText, tagColor
                        local aptTag = ownsApartment and " [APT]" or ""
                        local isDecayedCrosser = isIllegalCrosser and (wantedLevel == 0) and (tName == "Civilian")

                        if wantedLevel > 0 then
                            tagColor = CLR_ESP_RED
                            tagText = string.format("[%s] [*%d] %s (%d pts) - %dm%s", tName:upper(), stars, plr.Name, wantedLevel, dist, aptTag)
                        elseif isDecayedCrosser and isInsideUS then
                            tagColor = CLR_ESP_ORANGE
                            tagText = string.format("[%s] [SEARCH TABLET] %s (%dm)%s", tName:upper(), plr.Name, dist, aptTag)
                        elseif ownsApartment and aptLook then
                            tagColor = CLR_ESP_PURPLE
                            tagText = string.format("[%s] [APT] [SEARCH TABLET] %s (%dm)", tName:upper(), plr.Name, dist)
                        else
                            tagColor = teamColor
                            tagText = string.format("[%s] %s (%dm)%s", tName:upper(), plr.Name, dist, aptTag)
                        end

                        espList[#espList + 1] = { head = head, root = rootPart, tagText = tagText, color = tagColor }
                    end
                end
            end
        end

        cachedTargets = espList
        cachedSuspectPlayers = suspectData
        cachedSuspectRefs = suspectRefs
        cachedApartmentOwnerIds = aptOwnerIdSet
        task.wait(0.33)
    end
end)

-- ---------------------------------------------------------------------
-- RENDER & INTERACTION LOOP (Optimized: cached data, dirty panel pos)
-- ---------------------------------------------------------------------
RunService.RenderStepped:Connect(function()
    if not running then return end

    local mPos = getMousePos()
    local isMouseDown = isLeftMouseDown()

    local suspectData = cachedSuspectPlayers
    local totalSuspects = #suspectData
    local maxOffset = math.max(0, totalSuspects - MAX_ROWS)

    -- Mouse Click & Drag Logic
    if isMouseDown and not lastClickState then
        if isInside(windowPos, Vector2.new(windowSize.X, 32), mPos) then
            isDragging = true
            dragOffset = Vector2.new(mPos.X - windowPos.X, mPos.Y - windowPos.Y)
        else
            if isInside(btnScrollUpBg.Position, btnScrollUpBg.Size, mPos) then
                scrollOffset = math.clamp(scrollOffset - 1, 0, maxOffset)
            end
            if isInside(btnScrollDnBg.Position, btnScrollDnBg.Size, mPos) then
                scrollOffset = math.clamp(scrollOffset + 1, 0, maxOffset)
            end

            for i = 1, MAX_ROWS do
                local row = playerRows[i]
                if row.player and isInside(row.bg.Position, row.bg.Size, mPos) then
                    selectedPlayer = row.player
                    if setclipboard then
                        setclipboard(row.player.Name)
                        notify("Target Selected", "Selected & Copied '" .. row.player.Name .. "' for Tablet!", 2)
                    else
                        notify("Target Selected", "Selected: " .. row.player.Name, 2)
                    end
                    break
                end
            end

            if isInside(btnCopyBg.Position, btnCopyBg.Size, mPos) then
                if selectedPlayer then
                    if setclipboard then
                        setclipboard(selectedPlayer.Name)
                        notify("Tablet Helper", "Copied '" .. selectedPlayer.Name .. "'! Press Ctrl+V in Tablet.", 3)
                    end
                else
                    notify("Warning", "Click a suspect from the list first!", 2)
                end
            end

            if isInside(btnNearestBg.Position, btnNearestBg.Size, mPos) then
                local nearPlr, nearDist = getNearestUsSuspect()
                if nearPlr then
                    selectedPlayer = nearPlr
                    if setclipboard then setclipboard(nearPlr.Name) end
                    notify("Nearest Target", "Selected & Copied '" .. nearPlr.Name .. "' (" .. math.floor(nearDist) .. "m)!", 3)
                else
                    notify("Nearest Target", "No suspects found on US side!", 2)
                end
            end

            if isInside(btnUnstuckBg.Position, btnUnstuckBg.Size, mPos) then
                task.spawn(function()
                    local remotes = ReplicatedStorage:FindFirstChild("__remotes")
                    if remotes and remotes:FindFirstChild("VehicleService") and remotes.VehicleService:FindFirstChild("UnstuckVehicle") then
                        remotes.VehicleService.UnstuckVehicle:FireServer()
                        notify("Vehicle", "Vehicle Unstuck Request Sent!", 2)
                    end
                end)
            end

            if isInside(btnAptLookBg.Position, btnAptLookBg.Size, mPos) then
                apartmentLookEnabled = not apartmentLookEnabled
                scrollOffset = 0
                if apartmentLookEnabled then
                    notify("Apartment Look", "ENABLED - Showing apartment suspects on US side", 3)
                else
                    notify("Apartment Look", "DISABLED - Normal suspect view", 3)
                end
            end
        end
    elseif not isMouseDown then
        isDragging = false
    end

    if isDragging then
        windowPos = Vector2.new(mPos.X - dragOffset.X, mPos.Y - dragOffset.Y)
    end

    lastClickState = isMouseDown

    -- Only reposition panel elements when panel has moved (saves ~25 Drawing writes/frame)
    if windowPos.X ~= lastWindowPos.X or windowPos.Y ~= lastWindowPos.Y then
        repositionPanel()
        lastWindowPos = windowPos
    end

    -- Status label (only needs selectedPlayer distance - one cheap read)
    if selectedPlayer and selectedPlayer.Character and selectedPlayer.Character:FindFirstChild("HumanoidRootPart") then
        local myChar = lp.Character
        local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
        local dist = 0
        if myRoot then
            dist = math.floor((selectedPlayer.Character.HumanoidRootPart.Position - myRoot.Position).Magnitude)
        end
        statusLabel.Text = string.format("Target: %s (%dm) - Copied! Press Ctrl+V in Tablet", selectedPlayer.Name, dist)
        statusLabel.Color = CLR_STATUS_GOLD
    elseif selectedPlayer then
        statusLabel.Text = "Target: " .. selectedPlayer.Name .. " - Copied! Press Ctrl+V in Tablet"
        statusLabel.Color = CLR_STATUS_GOLD
    else
        statusLabel.Text = "Target: None (Click suspect row below to select & copy)"
        statusLabel.Color = CLR_STATUS_DIM
    end

    scrollOffset = math.clamp(scrollOffset, 0, maxOffset)
    local startRange = (totalSuspects > 0) and (scrollOffset + 1) or 0
    local endRange = math.min(scrollOffset + MAX_ROWS, totalSuspects)

    -- Update list range label
    if apartmentLookEnabled then
        listRangeLabel.Text = string.format("APT SUSPECTS US (%d-%d of %d)", startRange, endRange, totalSuspects)
        listRangeLabel.Color = CLR_LABEL_APT
    else
        listRangeLabel.Text = string.format("Suspects US (%d-%d of %d)", startRange, endRange, totalSuspects)
        listRangeLabel.Color = CLR_LABEL_NORMAL
    end

    -- Render Player Rows (uses pre-cached data - NO GetAttribute/FindFirstChild)
    local startY = windowPos.Y + 82
    local myChar = lp.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    local myPos = myRoot and myRoot.Position or Vector3.zero

    for i = 1, MAX_ROWS do
        local row = playerRows[i]
        local idx = i + scrollOffset
        local data = suspectData[idx]

        if data then
            local plr = data.plr
            row.player = plr

            local rPos = Vector2.new(windowPos.X + 8, startY + ((i - 1) * 26))
            row.bg.Position = rPos
            row.text.Position = Vector2.new(rPos.X + 8, rPos.Y + 5)

            -- All data is pre-cached from collector - zero attribute lookups
            local wantedLevel = data.wantedLevel
            local stars = data.stars
            local tName = data.tName
            local ownsApt = data.ownsApt

            -- Only distance needs live update (cheap Position read)
            local dist = 0
            local rootRef = data.rootRef
            if rootRef and myRoot then
                dist = math.floor((rootRef.Position - myPos).Magnitude)
            end

            local aptIndicator = ownsApt and " [APT]" or ""

            if plr == selectedPlayer then
                row.bg.Color = CLR_SELECTED_BG
                row.text.Color = CLR_SELECTED_TEXT
                if apartmentLookEnabled then
                    if wantedLevel > 0 then
                        row.text.Text = string.format("> [%s] %s (%dm) [APT]  |  *%d (%d pts)", tName:upper(), plr.Name, dist, stars, wantedLevel)
                    else
                        row.text.Text = string.format("> [%s] %s (%dm) [APT]  |  SEARCH TABLET", tName:upper(), plr.Name, dist)
                    end
                else
                    row.text.Text = string.format("> [%s] %s (%dm)%s  |  *%d (%d pts)", tName:upper(), plr.Name, dist, aptIndicator, stars, wantedLevel)
                end
            else
                if ownsApt and apartmentLookEnabled then
                    row.bg.Color = CLR_APT_ROW_BG
                else
                    row.bg.Color = CLR_ROW_BG
                end
                if apartmentLookEnabled then
                    if wantedLevel > 0 then
                        row.text.Color = CLR_APT_WANTED_TEXT
                        row.text.Text = string.format("  [%s] %s (%dm) [APT]  |  *%d (%d pts)", tName:upper(), plr.Name, dist, stars, wantedLevel)
                    else
                        row.text.Color = CLR_APT_SEARCH_TEXT
                        row.text.Text = string.format("  [%s] %s (%dm) [APT]  |  SEARCH TABLET", tName:upper(), plr.Name, dist)
                    end
                else
                    local teamColor = data.teamColor
                    if wantedLevel > 0 then
                        row.text.Color = CLR_WANTED_TEXT
                    else
                        row.text.Color = teamColor
                    end
                    row.text.Text = string.format("  [%s] %s (%dm)%s  |  *%d (%d pts)", tName:upper(), plr.Name, dist, aptIndicator, stars, wantedLevel)
                end
            end

            row.bg.Visible = true
            row.text.Visible = true
        else
            row.player = nil
            row.bg.Visible = false
            row.text.Visible = false
        end
    end

    -- APT LOOK button appearance
    if apartmentLookEnabled then
        btnAptLookBg.Color = CLR_APT_BTN_ON
        btnAptLookText.Text = "[APT LOOK: ON] - Filtering Apartment Suspects"
        btnAptLookText.Color = CLR_APT_BTN_TEXT_ON
    else
        btnAptLookBg.Color = CLR_APT_BTN_OFF
        btnAptLookText.Text = "[APT LOOK: OFF] - Show Apartment Suspects"
        btnAptLookText.Color = CLR_APT_BTN_TEXT_OFF
    end

    -- 3D ESP Rendering (only WorldToScreen + Drawing writes - no lookups)
    local currentTargets = cachedTargets
    local rendered = 0

    for i = 1, #currentTargets do
        if rendered >= POOL_SIZE then break end
        local t = currentTargets[i]

        local top, v1 = WorldToScreen(t.head.Position + Vector3.new(0, 0.7, 0))
        local bot, v2 = WorldToScreen(t.root.Position - Vector3.new(0, 3.2, 0))

        if v1 and v2 then
            rendered = rendered + 1
            local elem = pool[rendered]

            local h = math.max(4, bot.Y - top.Y)
            local w = math.max(3, h * 0.5)

            elem.box.Position = Vector2.new(top.X - w * 0.5, top.Y)
            elem.box.Size = Vector2.new(w, h)
            elem.box.Color = t.color

            elem.tag.Text = t.tagText
            elem.tag.Position = Vector2.new(top.X, top.Y - 14)

            elem.box.Visible = true
            elem.tag.Visible = true
        end
    end

    for i = rendered + 1, POOL_SIZE do
        pool[i].box.Visible = false
        pool[i].tag.Visible = false
    end
end)

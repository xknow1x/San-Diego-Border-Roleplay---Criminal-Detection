-- Roblox Physics Standard: 1 stud = 0.28 meters (approx 3.5714 studs per meter)
local STUDS_PER_METER = 3.5714

local function getDistanceMeters(pos1, pos2)
    if not pos1 or not pos2 then return 0 end
    local distStuds = (pos1 - pos2).Magnitude
    return math.max(0, math.floor(distStuds / STUDS_PER_METER))
end

end

local function getPlrParts(plr)
    if not plr or plr.Parent ~= Players then return nil, nil, nil end
    local char = plr.Character
    if not char then return nil, nil, nil end
    local root = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso") or char.PrimaryPart
    local head = char:FindFirstChild("Head")
    local hum  = char:FindFirstChildOfClass("Humanoid")
    local hp   = (hum and hum.Health) or 100
    if hp <= 0 then return nil, nil, nil end
    return root, head, hum
end

local BOAT_PROX_SQ = 35 * 35 -- 35 studs proximity (~10 meters)

local _boatsFolder = nil
local function getBoatsFolder()
    if _boatsFolder and _boatsFolder.Parent then return _boatsFolder end
    _boatsFolder = workspace:FindFirstChild("Boats")
    return _boatsFolder
end

local function isNearBoat(char, rootPart)
    if not rootPart then return false end
    local ok, result = pcall(function()
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if hum and hum.SeatPart then
            local seat = hum.SeatPart
            local p = seat.Parent
            while p and p ~= workspace do
                local name = p.Name:lower()
                if name:find("boat") or name:find("dinghy") or name:find("jetski") or p.Name == "Boats" then
                    return true
                end
                p = p.Parent
            end
        end

        local plrPos = rootPart.Position
        local folder = getBoatsFolder()
        if folder then
            for _, boat in ipairs(folder:GetChildren()) do
                local boatRoot = boat:FindFirstChild("Root") or boat.PrimaryPart or boat:FindFirstChildWhichIsA("BasePart")
                if boatRoot and boatRoot:IsA("BasePart") then
                    local diff = boatRoot.Position - plrPos
                    if (diff.X * diff.X + diff.Y * diff.Y + diff.Z * diff.Z) <= BOAT_PROX_SQ then
                        return true
                    end
                end
            end
        end

        if char and char.Parent and char.Parent ~= workspace then
            local pName = char.Parent.Name:lower()
            if pName:find("boat") or pName:find("dinghy") or pName:find("jetski") then
                return true
            end
        end
        return false
    end)
    return ok and result or false
end

local function getPlayerTeamName(plr)
    if not plr or plr.Parent ~= Players then return "Civilian" end
    local ok, name = pcall(function()
        if plr.Team then
            return plr.Team.Name
        end
        if plr.TeamColor then
            for _, t in ipairs(Teams:GetTeams()) do
                if t.TeamColor == plr.TeamColor then
                    return t.Name
                end
            end
        end
        return nil
    end)
    if ok and name then return name end
    return "Civilian"
end

local function isCivilian(plr)
    if not plr or plr.Parent ~= Players then return false end
    local tName = getPlayerTeamName(plr):lower()
    return tName == "civilian" or tName == "civilians" or tName:find("civilian") ~= nil
end

local function isLawEnforcement(plr)
    if not plr or plr.Parent ~= Players then return false end
    return not isCivilian(plr)
end

local function getWantedPoints(plr)
    if not plr or plr.Parent ~= Players then return 0 end
    local val = plr:GetAttribute("WantedLevel") or plr:GetAttribute("Wanted") or plr:GetAttribute("WantedPoints")
    if typeof(val) == "number" then return math.max(0, math.floor(val)) end
    local ls = plr:FindFirstChild("leaderstats")
    if ls then
        local w = ls:FindFirstChild("Wanted") or ls:FindFirstChild("Wanted Level") or ls:FindFirstChild("Bounty")
        if w and typeof(w.Value) == "number" then return math.max(0, math.floor(w.Value)) end
    end
    local wObj = plr:FindFirstChild("WantedLevel") or plr:FindFirstChild("Wanted")
    if wObj and typeof(wObj.Value) == "number" then return math.max(0, math.floor(wObj.Value)) end
    return 0
end

local function getWantedStars(points)
    if points <= 0 then return 0 end
    return math.clamp(math.ceil(points / 200), 1, 5)
end

local function isIllegalCrosser(plr)
    if not plr or plr.Parent ~= Players then return false end
    local fromMex = plr:GetAttribute("FromMexico") == true or plr:GetAttribute("IllegalCrosser") == true
    local legal = plr:GetAttribute("LegallyCrossedBorder") == true or plr:GetAttribute("LegallyCrossed") == true or plr:GetAttribute("Granted") == true
    return fromMex and not legal
end

-- BACKGROUND SCANNER (Refreshes player database every 0.2s)
-- =====================================================================
task.spawn(function()
    while running do
        pcall(function()
            local suspectData = {}
            local officerData = {}
            local aptOwnerIds = {}

            local playerList = Players:GetPlayers()

            for i = 1, #playerList do
                local plr = playerList[i]
                if plr ~= lp and plr.Parent == Players then
                    local aid = plr:GetAttribute("OwnedApartmentId")
                    if aid then aptOwnerIds[plr.UserId] = true end
                end
            end

            local aptLook  = apartmentLookEnabled
            local boatLook = boatSmugglingEnabled

            for i = 1, #playerList do
                local plr = playerList[i]
                if not plr or plr.Parent ~= Players or plr == lp then continue end

                local isCiv = isCivilian(plr)
                local isCop = isLawEnforcement(plr)
                local root, head, hum = getPlrParts(plr)

                local wantedLevel = getWantedPoints(plr)
                local stars       = getWantedStars(wantedLevel)
                local isIllegal   = isIllegalCrosser(plr)
                local ownsApt     = aptOwnerIds[plr.UserId] == true
                local isBoatSmug  = isCiv and root and isNearBoat(plr.Character, root) or false
                local isInUS      = root and (root.Position.X < BORDER_X) or false
                local tName       = getPlayerTeamName(plr)

                if isCop then
                    officerData[#officerData + 1] = {
                        plr         = plr,
                        tName       = tName,
                        isCop       = true,
                        isCiv       = false,
                        wantedLevel = 0,
                        stars       = 0,
                        isIllegal   = false,
                        ownsApt     = false,
                        isBoat      = false,
                        isInUS      = isInUS,
                    }
                elseif isCiv and isInUS then
                    local include = false
                    if boatLook then
                        include = isBoatSmug
                    elseif aptLook then
                        include = ownsApt and (wantedLevel > 0 or isIllegal)
                    else
                        include = wantedLevel > 0 or isIllegal or isBoatSmug
                    end

                    if include then

                            plr         = plr,
                            tName       = tName,
                            isCop       = false,
                            isCiv       = true,
                            wantedLevel = wantedLevel,

                            isIllegal   = isIllegal,
                            ownsApt     = ownsApt,
                            isBoat      = isBoatSmug,
                            isInUS      = isInUS,
                        }
                    end
                end
            end

            cachedAptOwners      = aptOwnerIds
            cachedOfficerPlayers = officerData
            cachedSuspectPlayers = suspectData
        end)

    end
